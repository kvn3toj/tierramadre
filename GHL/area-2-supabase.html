-- Bucket 'products': lectura pública, escritura solo admin/marketing
create policy "Public products read"
  on storage.objects for select
  using (bucket_id = 'products');

create policy "Admins write products"
  on storage.objects for all
  using (bucket_id = 'products' and (
    auth.jwt() ->> 'role' in ('super_admin','admin','marketing')
  ));

-- Bucket 'certificates': solo signed URLs, escritura admin
create policy "Admins write certificates"
  on storage.objects for all
  using (bucket_id = 'certificates' and (
    auth.jwt() ->> 'role' in ('super_admin','admin')
  ));

🛡️ Ejemplo de RLS Policies

-- ====== PRODUCTS · todos pueden leer activos, admin escribe ======
alter table products enable row level security;

create policy "Public read active products"
  on products for select
  using (activo = true);

create policy "Admins manage products"
  on products for all
  using (auth.jwt() ->> 'role' in ('super_admin','admin','marketing'));

-- ====== ORDERS · contact ve solo las suyas, admin ve todas ======
alter table orders enable row level security;

create policy "Contact reads own orders"
  on orders for select
  using (contact_id::text = auth.jwt() ->> 'contact_id');

create policy "Admins read all orders"
  on orders for select
  using (auth.jwt() ->> 'role' in ('super_admin','admin'));

-- ====== AMBASSADOR_LEADS · embajador ve solo los suyos ======
alter table ambassador_leads enable row level security;

create policy "Ambassador reads own leads"
  on ambassador_leads for select
  using (ambassador_id::text = (
    select id::text from ambassadors where user_id = auth.uid()
  ));

create policy "Ambassador creates own leads"
  on ambassador_leads for insert
  with check (ambassador_id::text = (
    select id::text from ambassadors where user_id = auth.uid()
  ));

-- ====== COMMISSIONS · embajador ve las suyas, super_admin gestiona ======
alter table commissions enable row level security;

create policy "Ambassador reads own commissions"
  on commissions for select
  using (ambassador_id::text = (
    select id::text from ambassadors where user_id = auth.uid()
  ));

create policy "Super admin manages commissions"
  on commissions for all
  using (auth.jwt() ->> 'role' = 'super_admin');

Edge Functions

Las 8 funciones serverless

Función 01

🔍 search-products

El bot llama aquí cuando un cliente describe lo que quiere. Devuelve top 3 productos según intent + presupuesto + ocasión.

Input: {intent, presupuesto, ocasion, ciudad}

Output: {productos: [...]}

Función 02

🛒 create-order

Recibe carrito + datos cliente, crea orden en DB con status pending, genera preferencia en Mercado Pago, devuelve URL de checkout.

Input: {contact, items, shipping}

Output: {order_id, mp_url}

Función 03

💳 mp-webhook

Recibe webhook de Mercado Pago. Valida firma. Actualiza orden a paid. Dispara workflow GHL #08 (post-venta).

Input: payload de MP

Output: 200 OK

Función 04

🎁 send-promo

Recibe promoción. Filtra audiencia por audience_filters. Para cada contacto: genera código único, envía mensaje por canal preferido (via GHL API).

Input: {promotion_id}

Output: {sent_count, errors}

Función 05

🏆 calculate-ambassador-score

Cron diario + trigger en eventos. Calcula score de cada embajador según fórmula Base 11. Actualiza nivel si cambia. Notifica.

Trigger: cron diario 00:00

Output: {updated_count}

Función 06

🎤 auto-event-invite

Al crear un evento, filtra contactos según audience_filters del evento y envía invitaciones automáticamente por canal preferido.

Input: {event_id}

Output: {invited_count}

Función 07

🔥 hot-lead-detector

B3. Cron cada 15 min. Evalúa señales en contactos activos. Si ≥4 señales y score >60, marca como hot_lead y notifica a Agente Senior.

Trigger: cron */15 * * * *

Output: {hot_leads_detected}

Función 08

🔄 ghl-sync

Bidireccional. Cuando cambia contact en Supabase → sync a GHL. Cuando llega webhook de GHL → sync a Supabase. Mantiene ambos sistemas alineados.

Trigger: DB trigger + webhook

Output: {synced: true}

📝 Implementación · invite-ambassador completa

// supabase/functions/invite-ambassador/index.ts
import { createClient } from 'jsr:@supabase/supabase-js@2';
import { serve } from "https://deno.land/std@0.224.0/http/server.ts";
import { Resend } from 'npm:resend';

interface InviteRequest {
  nombre_completo: string;
  email: string;
  celular: string;
  instagram_handle?: string;
  ciudad: string;
  nivel_inicial?: 'bronce' | 'plata' | 'oro' | 'diamante';
  comision_percent?: number;
  referido_por?: string;
  notas_internas?: string;
}

function slugify(text: string): string {
  return text
    .toLowerCase()
    .normalize('NFD').replace(/[̀-ͯ]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-|-$/g, '');
}

function randomCode(length = 4): string {
  return Math.random().toString(36).substring(2, 2 + length);
}

serve(async (req) => {
  if (req.method !== 'POST') return new Response('only POST', { status: 405 });

  try {
    const data: InviteRequest = await req.json();

    const supabase = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
    );

    // 1. Verificar que email no exista
    const { data: existing } = await supabase
      .from('ambassadors')
      .select('id')
      .eq('email', data.email)
      .single();

    if (existing) {
      return new Response(
        JSON.stringify({ error: 'Ya existe un embajador con ese correo' }),
        { status: 400, headers: { 'Content-Type': 'application/json' } }
      );
    }

    // 2. Generar slug único
    let slug = `${slugify(data.nombre_completo)}-${randomCode()}`;
    let slugExists = true;
    while (slugExists) {
      const { data: check } = await supabase
        .from('ambassadors')
        .select('id')
        .eq('slug', slug)
        .single();
      if (!check) slugExists = false;
      else slug = `${slugify(data.nombre_completo)}-${randomCode()}`;
    }

    // 3. Crear usuario Auth con rol embajador
    const { data: authUser, error: authErr } = await supabase.auth.admin.inviteUserByEmail(
      data.email,
      {
        data: {
          role: 'embajador',
          nombre_completo: data.nombre_completo,
          slug
        },
        redirectTo: `${Deno.env.get('FRONTEND_URL')}/embajador/onboarding`
      }
    );

    if (authErr) throw authErr;

    // 4. Determinar % comisión según nivel
    const comisionPorNivel: Record<string, number> = {
      bronce: 8, plata: 10, oro: 12, diamante: 15
    };
    const comision = data.comision_percent ?? comisionPorNivel[data.nivel_inicial ?? 'bronce'];

    // 5. INSERT en tabla ambassadors
    const { data: ambassador, error: insErr } = await supabase
      .from('ambassadors')
      .insert({
        user_id: authUser.user!.id,
        full_name: data.nombre_completo,
        email: data.email,
        celular: data.celular,
        instagram_handle: data.instagram_handle,
        ciudad: data.ciudad,
        slug,
        nivel: data.nivel_inicial ?? 'bronce',
        comision_percent: comision,
        status: 'invited',
        referido_por: data.referido_por,
        notas: data.notas_internas
      })
      .select()
      .single();

    if (insErr) throw insErr;

    // 6. Crear carpeta Storage para la embajadora
    await supabase.storage
      .from('ambassadors')
      .upload(`${ambassador.id}/.keep`, new Blob([''], { type: 'text/plain' }));

    // 7. Sincronizar tag a GHL (vía Cloudflare Worker)
    await fetch(`${Deno.env.get('CF_WORKER_URL')}/sync-ambassador-tag`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${Deno.env.get('INTERNAL_API_SECRET')}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        ambassador_id: ambassador.id,
        slug,
        action: 'create'
      })
    });

    // 8. Enviar email custom de bienvenida con Resend
    const resend = new Resend(Deno.env.get('RESEND_API_KEY'));
    const formUrl = `${Deno.env.get('FRONTEND_URL')}/embajador/${slug}/leads`;
    const dashboardUrl = `${Deno.env.get('FRONTEND_URL')}/embajador`;

    await resend.emails.send({
      from: 'Tierra Madre <hola@tierramadre.co>',
      to: data.email,
      subject: `¡Bienvenida al equipo, ${data.nombre_completo.split(' ')[0]}!`,
      html: getWelcomeTemplate({
        nombre: data.nombre_completo.split(' ')[0],
        formUrl,
        dashboardUrl,
        nivel: data.nivel_inicial ?? 'bronce',
        comision
      })
    });

    return new Response(JSON.stringify({
      id: ambassador.id,
      slug,
      form_url: formUrl,
      dashboard_url: dashboardUrl,
      email_sent: true
    }), {
      headers: { 'Content-Type': 'application/json' }
    });

  } catch (err) {
    return new Response(JSON.stringify({ error: err.message }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' }
    });
  }
});

<html><body style="font-family:Inter,sans-serif;background:#0f1714;color:#e6f4ef;padding:40px">
  <div style="max-width:600px;margin:0 auto">

    <p>Oficialmente eres parte del equipo de Tierra Madre.</p>

// supabase/functions/search-products/index.ts
import { createClient } from 'jsr:@supabase/supabase-js@2';
import { serve } from "https://deno.land/std@0.224.0/http/server.ts";

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders });

  try {
    const { intent, presupuesto, ocasion, ciudad } = await req.json();

    const supabase = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
    );

    // Margen 20% sobre presupuesto para no perder ventas cercanas
    const precio_max = presupuesto ? presupuesto * 1.2 : 99999999;

    let query = supabase
      .from('products')
      .select(`
        id, sku, nombre, slug, descripcion_corta,
        precio_cop, certificado_url,
        product_images!inner(url, is_primary)
      `)
      .eq('activo', true)
      .gt('stock', 0)
      .lte('precio_cop', precio_max);

    if (intent?.categoria) query = query.eq('categoria', intent.categoria);
    if (ocasion) query = query.contains('ocasion', [ocasion]);

    const { data: productos, error } = await query
      .order('destacado', { ascending: false })
      .order('precio_cop', { ascending: false })
      .limit(3);

    if (error) throw error;

    return new Response(JSON.stringify({ productos }), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' }
    });
  } catch (err) {
    return new Response(JSON.stringify({ error: err.message }), {
      status: 500,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' }
    });
  }
});
